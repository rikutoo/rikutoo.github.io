\section*{Job Market Paper}
\begin{cvsection}
  ``Exchange Rate Disconnect with Disagreeing Arbitrageurs''
  \begin{projectdetails}
    \textbf{Abstract:} I develop a two-country general equilibrium model in which international arbitrageurs disagree about future exchange rates. Trading gains and losses redistribute wealth across arbitrageurs, changing whose beliefs determine equilibrium prices. This mechanism amplifies exchange rate responses to standard macroeconomic shocks and generates an endogenous wedge in uncovered interest parity, without requiring a separate financial shock. Calibrated to macroeconomic, financial, and survey data, the model accounts for exchange rates’ high volatility and weak relationship with observable fundamentals. The calibrated wealth-share multiplier implies that amplification accounts for approximately 70 percent of exchange rate variance.
  \end{projectdetails}
\end{cvsection}

\section*{Work in Progress}
\begin{cvsection}
  \project{Stock Market Uncertainty and Firm Matching}{with Husang Kim and Kai Wu}
  \begin{projectdetails}
    Uses the universe of Chinese VAT invoices and exploits the 2015 stock market
    turbulence as a natural experiment. Documents a contraction in B2B transactions
    for firms more exposed to stock market uncertainty. Develops a frictional
    matching model with stock prices as noisy signals.
  \end{projectdetails}

  \project{Disagreement, Monetary Policy, and Term Premia}
    {with Stefano Eusepi and Bruce Preston}
  \begin{projectdetails}
    Develops a New Keynesian model with heterogeneous beliefs and short/long bonds.
    Explores stability and determinacy in the presence of heterogeneous beliefs
    and aims to capture time-varying term premia.
  \end{projectdetails}
