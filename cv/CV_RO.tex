\section*{Work in Progress}

\project{Exchange Rate Disconnect with Disagreeing Arbitrageurs}{\textit{Job Market Paper}}
\vspace{4pt}
I investigate the disconnect between exchange rates and macroeconomic fundamentals through
the lens of heterogeneous beliefs. I develop a two-country general equilibrium model in which
international arbitrageurs disagree about future exchange-rate dynamics. Their disagreement
generates a novel endogenous deviation from uncovered interest parity (UIP), which I call the
wealth-share disagreement wedge. Calibrated using survey data on exchange rate expectations,
the model quantitatively accounts for several key dimensions of the macro--exchange disconnect.

\project{Stock Market Uncertainty and Firm Matching}{Joint with Husang Kim and Kai Wu}
\begin{researchpoints}
  \item Uses universe of Chinese VAT invoices data and exploits the 2015 stock market turbulence
        as a natural experiment
  \item Documents contraction in B2B transactions for firms that were more exposed to stock
        market uncertainty
  \item Develops a frictional matching model with stock prices as noisy signals
\end{researchpoints}

\project{Disagreement, Monetary Policy, and Term Premia}{Joint with Stefano Eusepi and Bruce Preston}
\begin{researchpoints}
  \item New Keynesian model with heterogeneous beliefs and short/long bonds
  \item Explores stability and determinacy of the model in the presence of heterogeneous beliefs
  \item Aims to capture time-varying term premia
\end{researchpoints}
